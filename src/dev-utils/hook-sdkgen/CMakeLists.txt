cmake_minimum_required(VERSION 3.13)
add_executable(hook-sdkgen
    src/main.cpp
    src/model.cpp
    src/pe_image.cpp
    src/elf_image.cpp
    src/dwarf_reader.cpp
    src/hash_util.cpp
    src/zydis_validator.cpp
    src/manifest_emitter.cpp
    src/binary_emitter.cpp
    src/report.cpp
    src/reader_factory.cpp
    src/dia_reader.cpp
    src/sdk_emitter.cpp
)
target_include_directories(hook-sdkgen PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)
target_compile_features(hook-sdkgen PRIVATE cxx_std_17)
set_target_properties(hook-sdkgen PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_definitions(hook-sdkgen PRIVATE ORCA_TEMPLATE_DIR="${CMAKE_SOURCE_DIR}/sdk/plugin_v1/templates")
if (WIN32 AND MSVC)
    # DIA SDK discovery: prefer VSINSTALLDIR/DIA SDK, optionally search current VS installs
    set(_hook_dia_found FALSE)
    set(_hook_dia_inc "")
    set(_hook_dia_lib "")
    if(DEFINED ENV{VSINSTALLDIR} AND EXISTS "$ENV{VSINSTALLDIR}DIA SDK/include/dia2.h")
        set(_hook_dia_inc "$ENV{VSINSTALLDIR}DIA SDK/include")
        set(_hook_dia_lib "$ENV{VSINSTALLDIR}DIA SDK/lib/amd64/diaguids.lib")
        if(EXISTS "${_hook_dia_inc}/dia2.h" AND EXISTS "${_hook_dia_lib}")
            set(_hook_dia_found TRUE)
        endif()
    endif()
    if(NOT _hook_dia_found)
        # Fallback: search common VS install locations (handles vcvars not run)
        file(GLOB _hook_vs_dia_paths
            "C:/Program Files/Microsoft Visual Studio/*/Community/DIA SDK/include/dia2.h"
            "C:/Program Files/Microsoft Visual Studio/*/Professional/DIA SDK/include/dia2.h"
            "C:/Program Files/Microsoft Visual Studio/*/Enterprise/DIA SDK/include/dia2.h"
            "C:/Program Files/Microsoft Visual Studio/*/BuildTools/DIA SDK/include/dia2.h"
        )
        foreach(_hook_dia_file IN LISTS _hook_vs_dia_paths)
            if(EXISTS "${_hook_dia_file}")
                get_filename_component(_hook_dia_inc "${_hook_dia_file}" DIRECTORY)
                get_filename_component(_hook_dia_sdk_dir "${_hook_dia_inc}" DIRECTORY)
                set(_hook_dia_lib "${_hook_dia_sdk_dir}/lib/amd64/diaguids.lib")
                if(EXISTS "${_hook_dia_lib}")
                    set(_hook_dia_found TRUE)
                    break()
                endif()
            endif()
        endforeach()
    endif()
    if(_hook_dia_found)
        target_include_directories(hook-sdkgen PRIVATE "${_hook_dia_inc}")
        target_link_libraries(hook-sdkgen PRIVATE "${_hook_dia_lib}")
        target_compile_definitions(hook-sdkgen PRIVATE ORCA_DIA_AVAILABLE=1)
        message(STATUS "hook-sdkgen: DIA SDK found at ${_hook_dia_inc} (ORCA_DIA_AVAILABLE=1)")
        # Locate msdia140.dll for reg-free LoadLibrary (no registration required)
        if(NOT DEFINED _hook_dia_sdk_dir)
            get_filename_component(_hook_dia_sdk_dir "${_hook_dia_inc}" DIRECTORY)
        endif()
        set(_hook_dia_dll "${_hook_dia_sdk_dir}/bin/amd64/msdia140.dll")
        if(NOT EXISTS "${_hook_dia_dll}")
            set(_hook_dia_dll "${_hook_dia_sdk_dir}/bin/msdia140.dll")
        endif()
        if(NOT EXISTS "${_hook_dia_dll}")
            file(GLOB _hook_dia_dll_candidates "${_hook_dia_sdk_dir}/bin/*/msdia*.dll")
            if(_hook_dia_dll_candidates)
                list(GET _hook_dia_dll_candidates 0 _hook_dia_dll)
            endif()
        endif()
        if(EXISTS "${_hook_dia_dll}")
            file(TO_CMAKE_PATH "${_hook_dia_dll}" _hook_dia_dll_cmake)
            target_compile_definitions(hook-sdkgen PRIVATE "ORCA_DIA_DLL_PATH=\"${_hook_dia_dll_cmake}\"")
            message(STATUS "hook-sdkgen: DIA DLL at ${_hook_dia_dll} (ORCA_DIA_DLL_PATH)")
        else()
            message(STATUS "hook-sdkgen: DIA DLL not found under ${_hook_dia_sdk_dir}/bin — runtime will try VSINSTALLDIR and exe dir")
        endif()
    else()
        if(ORCA_GENERATE_HOOK_SDK)
            message(FATAL_ERROR "hook-sdkgen: DIA SDK not found (dia2.h + diaguids.lib) under VSINSTALLDIR/DIA SDK or VS installs. Run vcvars64.bat or install DIA SDK before configuring with ORCA_GENERATE_HOOK_SDK=ON on Windows.")
        else()
            message(STATUS "hook-sdkgen: DIA SDK not found — compiling stub (ORCA_DIA_AVAILABLE not defined, SDK OFF)")
        endif()
    endif()
    target_link_libraries(hook-sdkgen PRIVATE ole32 oleaut32)
    # Zydis from SafetyHook amalgamated dep — use installed files, not find_package
    # When generation is ON, missing files is FATAL; remove QUIET fallback.
    if(ORCA_GENERATE_HOOK_SDK)
        set(_hook_zydis_h "${CMAKE_PREFIX_PATH}/include/safetyhook/Zydis.h")
        set(_hook_zydis_c "${CMAKE_PREFIX_PATH}/include/safetyhook/Zydis.c")
        if(NOT EXISTS "${_hook_zydis_h}" AND EXISTS "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.h")
            set(_hook_zydis_h "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.h")
        endif()
        if(NOT EXISTS "${_hook_zydis_c}" AND EXISTS "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.c")
            set(_hook_zydis_c "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.c")
        endif()
        if(NOT EXISTS "${_hook_zydis_h}" OR NOT EXISTS "${_hook_zydis_c}")
            message(FATAL_ERROR "Zydis amalgamated files not found at ${_hook_zydis_h} / ${_hook_zydis_c} (CMAKE_PREFIX_PATH=${CMAKE_PREFIX_PATH}). Build deps with dep_SafetyHook (SafetyHook v0.7.0 amalgamated-zydis, SHA 505d4c07ec1c5b94a17f3906ca86afbe1264e738d8becaa244866694c6200c2c) before configuring with ORCA_GENERATE_HOOK_SDK=ON.")
        endif()
        target_include_directories(hook-sdkgen PRIVATE "${CMAKE_PREFIX_PATH}/include" "${CMAKE_PREFIX_PATH}/include/safetyhook")
        set_source_files_properties("${_hook_zydis_c}" PROPERTIES LANGUAGE C)
        target_sources(hook-sdkgen PRIVATE "${_hook_zydis_c}")
        target_compile_definitions(hook-sdkgen PRIVATE HAVE_ZYDIS=1 ZYDIS_STATIC_BUILD)
        message(STATUS "hook-sdkgen: Zydis amalgamated found at ${_hook_zydis_h}, instruction validation enabled")
    else()
        # Generation OFF — hook-sdkgen still builds but Zydis is optional and not required
        # Do not search for system Zydis; validation will be disabled with reason at runtime.
        message(STATUS "hook-sdkgen: ORCA_GENERATE_HOOK_SDK=OFF — Zydis not required")
    endif()
else()
    # Non-Windows: same Zydis handling (no DIA)
    if(ORCA_GENERATE_HOOK_SDK)
        set(_hook_zydis_h "${CMAKE_PREFIX_PATH}/include/safetyhook/Zydis.h")
        set(_hook_zydis_c "${CMAKE_PREFIX_PATH}/include/safetyhook/Zydis.c")
        if(NOT EXISTS "${_hook_zydis_h}" AND EXISTS "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.h")
            set(_hook_zydis_h "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.h")
        endif()
        if(NOT EXISTS "${_hook_zydis_c}" AND EXISTS "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.c")
            set(_hook_zydis_c "${CMAKE_PREFIX_PATH}/include/safetyhook/zydis.c")
        endif()
        if(NOT EXISTS "${_hook_zydis_h}" OR NOT EXISTS "${_hook_zydis_c}")
            message(FATAL_ERROR "Zydis amalgamated files not found at ${_hook_zydis_h} / ${_hook_zydis_c} (CMAKE_PREFIX_PATH=${CMAKE_PREFIX_PATH}). Build deps with dep_SafetyHook before configuring with ORCA_GENERATE_HOOK_SDK=ON.")
        endif()
        target_include_directories(hook-sdkgen PRIVATE "${CMAKE_PREFIX_PATH}/include" "${CMAKE_PREFIX_PATH}/include/safetyhook")
        set_source_files_properties("${_hook_zydis_c}" PROPERTIES LANGUAGE C)
        target_sources(hook-sdkgen PRIVATE "${_hook_zydis_c}")
        target_compile_definitions(hook-sdkgen PRIVATE HAVE_ZYDIS=1 ZYDIS_STATIC_BUILD)
        message(STATUS "hook-sdkgen: Zydis amalgamated found (non-Windows)")
    endif()
endif()
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
    target_link_libraries(hook-sdkgen PRIVATE ZLIB::ZLIB)
    target_compile_definitions(hook-sdkgen PRIVATE HAVE_ZLIB=1)
    message(STATUS "hook-sdkgen: ZLIB found — manifest compression enabled")
else()
    message(STATUS "hook-sdkgen: ZLIB not found — manifest will be written uncompressed only")
endif()

endif()
